// ---------------- GAME 1: PICTIONARY ----------------

function PictionaryGame() {

  const [currentIdx, setCurrentIdx] = useState(0);

  const [feedback, setFeedback] = useState(null); // 'correct' | 'wrong'

  const [score, setScore] = useState(0);


  const item = PICTIONARY_ITEMS[currentIdx];


  const handleGuess = (option) => {

    if (feedback) return;

    if (option === item.correct) {

      setFeedback('correct');

      setScore(s => s + 1);

      setTimeout(() => {

        setFeedback(null);

        setCurrentIdx(prev => (prev + 1) % PICTIONARY_ITEMS.length);

      }, 1500);

    } else {

      setFeedback('wrong');

      setTimeout(() => setFeedback(null), 1000);

    }

  };


  return (

    <div className="flex flex-col items-center justify-center h-full p-8 text-center">

      <h2 className="text-2xl font-bold text-blue-400 mb-8 flex items-center gap-2">

        <Image /> Pictionary

      </h2>


      <div className="bg-slate-700 p-8 rounded-3xl mb-8 shadow-2xl border-4 border-slate-600 animate-in zoom-in">

        <div className="text-9xl mb-4 transform hover:scale-110 transition-transform cursor-pointer">

          {item.icon}

        </div>

        <p className="text-slate-300 font-bold text-lg">{item.question}</p>

      </div>


      <div className="grid grid-cols-1 md:grid-cols-3 gap-4 w-full max-w-3xl">

        {item.options.map((opt, i) => (

          <button

            key={i}

            onClick={() => handleGuess(opt)}

            disabled={feedback !== null}

            className={`p-4 rounded-xl font-bold text-lg border-2 transition-all

              ${feedback === 'correct' && opt === item.correct 

                ? 'bg-emerald-600 border-emerald-400 text-white' 

                : feedback === 'wrong' 

                  ? 'bg-slate-800 border-slate-600 opacity-50' 

                  : 'bg-slate-800 border-slate-600 hover:border-blue-400 hover:bg-slate-700'}`}

          >

            {opt}

          </button>

        ))}

      </div>

      

      {feedback === 'correct' && <div className="mt-6 text-emerald-400 text-xl font-bold animate-bounce">Doğru Bildin! 🎉</div>}

      {feedback === 'wrong' && <div className="mt-6 text-rose-500 text-xl font-bold">Yanlış, tekrar dene. ❌</div>}

    </div>

  );

}

// ---------------- GAME 5: JIGSAW ----------------

function JigsawGame() {

  const [currentIdx, setCurrentIdx] = useState(0);

  const [pieces, setPieces] = useState([]);

  const [userOrder, setUserOrder] = useState([]);

  const [isCorrect, setIsCorrect] = useState(null);


  const sentenceData = JIGSAW_SENTENCES[currentIdx];


  useEffect(() => {

    // Shuffle pieces

    const shuffled = [...sentenceData.parts].sort(() => Math.random() - 0.5);

    setPieces(shuffled);

    setUserOrder([]);

    setIsCorrect(null);

  }, [currentIdx]);


  const addToLine = (word) => {

    setUserOrder([...userOrder, word]);

    setPieces(pieces.filter(p => p !== word)); // Simple filter (assumes unique words for demo)

  };


  const removeFromLine = (word) => {

    setPieces([...pieces, word]);

    setUserOrder(userOrder.filter(w => w !== word));

    setIsCorrect(null);

  };


  const check = () => {

    if (userOrder.join(' ') === sentenceData.correct) {

      setIsCorrect(true);

    } else {

      setIsCorrect(false);

    }

  };


  const next = () => {

    setCurrentIdx(prev => (prev + 1) % JIGSAW_SENTENCES.length);

  };


  return (

    <div className="flex flex-col items-center justify-center h-full p-8 text-center">

      <h2 className="text-2xl font-bold text-violet-400 mb-8 flex items-center gap-2">

        <Move /> Sentence Jigsaw

      </h2>


      {/* Drop Zone */}

      <div className={`w-full max-w-3xl min-h-[100px] bg-slate-900 rounded-2xl border-2 border-dashed flex flex-wrap gap-2 p-6 mb-8 items-center justify-center transition-all

        ${isCorrect === true ? 'border-emerald-500 shadow-[0_0_30px_rgba(16,185,129,0.2)]' : 'border-slate-600'}`}>

        {userOrder.length === 0 && <span className="text-slate-600">Kelimeleri buraya taşı...</span>}

        {userOrder.map((word, i) => (

          <button key={i} onClick={() => removeFromLine(word)} className="bg-violet-600 hover:bg-violet-500 px-4 py-2 rounded-lg text-white font-bold animate-in zoom-in">

            {word}

          </button>

        ))}

      </div>


      {/* Pieces */}

      <div className="flex flex-wrap gap-3 justify-center max-w-2xl mb-12">

        {pieces.map((word, i) => (

          <button 

            key={i} 

            onClick={() => addToLine(word)}

            className="bg-slate-700 hover:bg-slate-600 px-4 py-2 rounded-lg text-slate-200 font-medium border border-slate-600 transition-all hover:-translate-y-1"

          >

            {word}

          </button>

        ))}

      </div>


      <div className="flex gap-4">

        {isCorrect !== true && (

          <button onClick={check} className="px-8 py-3 bg-slate-100 text-slate-900 rounded-full font-bold hover:scale-105 transition-all">

            Kontrol Et

          </button>

        )}

        {isCorrect === true && (

          <button onClick={next} className="px-8 py-3 bg-emerald-500 text-white rounded-full font-bold hover:scale-105 transition-all flex items-center gap-2">

            Sonraki <ArrowRight size={20}/>

          </button>

        )}

      </div>

      

      {isCorrect === false && <p className="mt-4 text-rose-500 font-bold">Sıralama yanlış, tekrar dene.</p>}

    </div>

  );

}
